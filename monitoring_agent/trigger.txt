CREATE OR REPLACE FUNCTION notify_email_insert()
RETURNS TRIGGER AS $$
BEGIN
    PERFORM pg_notify('new_email', json_build_object('id', NEW.id, 'email', NEW.email, 'tenant_id', NEW.tenant_id)::text);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_email_insert
AFTER INSERT ON emails
FOR EACH ROW EXECUTE FUNCTION notify_email_insert();
